% -------- FACTS --------
male(john).
male(paul).
male(mike).

female(lisa).
female(susan).

parent(john, paul).
parent(john, lisa).
parent(paul, mike).
parent(lisa, susan).

% -------- RULES --------
father(X, Y) :-
    male(X),
    parent(X, Y).

mother(X, Y) :-
    female(X),
    parent(X, Y).

grandparent(X, Y) :-
    parent(X, Z),
    parent(Z, Y).

sibling(X, Y) :-
    parent(Z, X),
    parent(Z, Y),
    X \= Y.

% -------- GOAL / QUERIES --------
% Example queries you can run in Prolog:

% ?- father(john, paul).
% ?- mother(lisa, susan).
% ?- grandparent(john, mike).
% ?- sibling(paul, lisa).

type: swipl
Save file as family.pl
Open Prolog and load: ?- consult('family.pl').
Run queries: ?- father(john, paul).
exit: halt
